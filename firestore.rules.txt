rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isValidOrder(data) {
      return data.keys().hasAll(['orderNumber', 'status'])
        && data.orderNumber is string
        && data.orderNumber.size() <= 40
        && data.status in ['preparing', 'cooking', 'ready', 'done', 'rejected']
        && (!('total' in data) || (data.total is number && data.total >= 0 && data.total <= 500000));
    }
    match /orders/{orderId} {
      allow read: if true;
      allow create, update: if isValidOrder(request.resource.data);
      allow delete: if false;
    }
    match /coffee_people_orders/{orderId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().hasAll(['jsonPayload'])
        && request.resource.data.jsonPayload is string
        && request.resource.data.jsonPayload.size() < 60000;
      allow delete: if false;
    }
    match /coffee_people_users/{userId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().hasAll(['jsonPayload'])
        && request.resource.data.jsonPayload is string
        && request.resource.data.jsonPayload.size() < 60000;
      allow delete: if false;
    }
  }
}
