rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isValidOrder(data) {
      return data.keys().hasAll(['clientName', 'items', 'total', 'status'])
        && data.clientName is string
        && data.clientName.size() >= 1
        && data.clientName.size() <= 60
        && data.items is list
        && data.items.size() >= 1
        && data.items.size() <= 50
        && data.total is number
        && data.total >= 0
        && data.total <= 500000
        && data.status in ['new', 'preparing', 'cooking', 'ready', 'done'];
    }

    function isValidStatusUpdate(newData, oldData) {
      return newData.diff(oldData).affectedKeys().hasOnly(['status', 'updatedAt'])
        && newData.status in ['new', 'preparing', 'cooking', 'ready', 'done'];
    }

    match /orders/{orderId} {
      allow read: if true;
      allow create: if isValidOrder(request.resource.data);
      allow update: if isValidStatusUpdate(request.resource.data, resource.data)
                    || isValidOrder(request.resource.data);
      allow delete: if false;
    }

    match /coffee_people_orders/{orderId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().hasAll(['jsonPayload', 'updatedAt'])
                            && request.resource.data.jsonPayload is string
                            && request.resource.data.jsonPayload.size() <= 15000;
      allow delete: if false;
    }

    match /coffee_people_users/{userId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().hasAll(['jsonPayload', 'updatedAt'])
                            && request.resource.data.jsonPayload is string
                            && request.resource.data.jsonPayload.size() <= 15000;
      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
